//ImageTypes
(FixedInternalImagePixelType "float")
(FixedImageDimension 3)
(MovingInternalImagePixelType "float")
(MovingImageDimension 3)

//Components
// CHANGED: MultiResolutionRegistration -> MultiMetricMultiResolutionRegistration
// so we can combine the image metric with a smoothness penalty.
(Registration "MultiMetricMultiResolutionRegistration")
(FixedImagePyramid "FixedSmoothingImagePyramid")
(MovingImagePyramid "MovingSmoothingImagePyramid")
(Interpolator "BSplineInterpolator")

// CHANGED: added TransformBendingEnergyPenalty as a second metric.
// This is the regularization that was missing and let the field fold.
(Metric "AdvancedMattesMutualInformation" "TransformBendingEnergyPenalty")
(Metric0Weight 1.0)
// CHANGED: 1.0 -> 1000. Logged Metric1 (bending energy) values are ~1e-4,
// vs Metric0 (MI) ~0.1-0.3 — a ~1000-3000x gap. At weight 1.0 the penalty
// was numerically inert (contributed nothing to the optimized total), so
// the field was never actually regularized. Raise further (5000, 10000) if
// the field still folds; lower if the deformation looks over-rigid.
(Metric1Weight 1000)

// CHANGED: StandardGradientDescent (fixed huge step) -> ASGD (auto step size)
(Optimizer "AdaptiveStochasticGradientDescent")
(AutomaticParameterEstimation "true")

(ResampleInterpolator "FinalBSplineInterpolator")
(Resampler "DefaultResampler")
(Transform "BSplineTransform")

(ErodeMask "false" )

(NumberOfResolutions 3)
(FinalGridSpacingInVoxels 25.000000 25.000000 25.000000)

(HowToCombineTransforms "Compose")

(WriteTransformParametersEachIteration "false")
(WriteResultImage "true")
(CompressResultImage "false")
(WriteResultImageAfterEachResolution "false")
// CHANGED: "false" -> "true". With RandomCoordinate sampling the per-
// iteration "Final metric value" in the log is just one noisy stochastic
// sample, not comparable across runs. This makes elastix compute the real
// full-image metric at the end of each resolution instead.
(ShowExactMetricValue "true")
(WriteDiffusionFiles "true")

// Option supported in elastix 4.1:
(UseFastAndLowMemoryVersion "true")

//Maximum number of iterations in each resolution level:
// CHANGED: 5000 -> 2000. With ASGD you no longer need to burn 5000 steps;
// convergence should be visible well before this.
(MaximumNumberOfIterations 2000 )

//Number of grey level bins in each resolution level:
(NumberOfHistogramBins 128 )
(FixedLimitRangeRatio 0.0)
(MovingLimitRangeRatio 0.0)
(FixedKernelBSplineOrder 3)
(MovingKernelBSplineOrder 3)

//Number of spatial samples used to compute the mutual information in each resolution level:
(ImageSampler "RandomCoordinate")
(FixedImageBSplineInterpolationOrder 1 )
// CHANGED: "true" -> "false" (and removed the now-unused SampleRegionSize).
// UseRandomSampleRegion=true was silently disabling ASGD's
// UseAdaptiveStepSizes (elastix logs this warning explicitly), which
// defeated part of the point of switching to ASGD in the first place.
(UseRandomSampleRegion "false")
// CHANGED: 10000 -> 3000 per iteration is fine for ASGD; you can go back up
// if gradients look noisy. (More samples = slower but smoother gradient.)
(NumberOfSpatialSamples 3000 )
(NewSamplesEveryIteration "true")
(CheckNumberOfSamples "true")
(MaximumNumberOfSamplingAttempts 8)

//Order of B-Spline interpolation used in each resolution level:
(BSplineInterpolationOrder 3)

//Order of B-Spline interpolation used for applying the final deformation:
(FinalBSplineInterpolationOrder 3)

//Default pixel value for pixels that come from outside the picture:
(DefaultPixelValue 0)

// REMOVED: manual step-size schedule (SP_a / SP_A / SP_alpha).
// ASGD with AutomaticParameterEstimation estimates these itself; leaving them
// in would fight the auto-estimation, which was the whole problem before.
